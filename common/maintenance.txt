        $('.common-content').removeClass(function (_, cls) {
            return cls.replace('common-content', '').trim();
        });

        const parts = document.title.split('|')[0]
            .split('<').map(s => $.trim(s)).filter(Boolean);

        if (parts.length === 2) {
            $('.common-content .tab-page-scroll.js-scroll-tab:eq(0)').hide();
        }

        const depthText = parts.length > 1 ? parts[parts.length - 2] : parts[0] || '';

        $('.content-main-title').text(depthText);

        $('.common-page-title').html(`
            <h2 class="title">${depthText}</h2>
            <button type="button" onclick="history.back();" class="btn-back">Back</button>
        `);

        $('.common-content').each(function () {
            const $container = $(this);

            function hideIfMultiple(targetClass, indexToHide) {
                const $targets = $container.find(targetClass);
                if ($targets.length > 1) {
                    $targets.eq(indexToHide).hide();
                }
            }

            const hasSnowparkParent = $container.closest('.section-snowpark.site-pyeongchang').length > 0;

            if (!hasSnowparkParent) {
                hideIfMultiple('.tab-page-scroll.js-scroll-tab', 0);
                hideIfMultiple('.common-top-info', 0);
                hideIfMultiple('.common-photo-slider', 0);
            }

            hideIfMultiple('.foot-sticky-action', 1);
        });

        /* ===== 성수기 ===== */
        const dday = Math.ceil(
            (new Date('2026-05-13') - new Date()) / (1000 * 60 * 60 * 24)
        );

        $('#dDay').text(dday > 0 ? 'D-' + dday : 'D-Day');

        /* ===============================================
        data layout table - add column for 3 column layout
        =============================================== */
        $('.data-layout-table').each(function () {
            const $table = $(this);

            if ($table.find('tbody tr').filter(function () {
                return $(this).children('td').length === 2;
            }).length) {
                $table.find('colgroup').append('<col style="width:0;">');
            }
        });

    });
</script>
<script>
$(document).one('ajaxStop', function () {

    if (!$('body').hasClass('site-pyeongchang')) return;

    setTimeout(function () {

        $('*').each(function () {
            if ($(this).children().length) return;

            const text = $.trim($(this).text());

            if (
                /^생수/i.test(text) ||
                /^Bottled water/i.test(text) ||
                /^1L Bottled water/i.test(text)
            ) {
                $(this).remove();
                return;
            }

            $(this).text(
                text
                    .replace('/생수', '')
                    .replace('/Bottled water', '')
            );
        });

    }, 1000);

});
</script>
